    const t=nearTrainer();
    if(t){

        dialog('<b>Maître du Cœur</b><br>Tu n’as pas encore vaincu les dix Gardiens du Sanctuaire.');
        return
      }

      else dialog('<b>'+t.dialog+'</b><br>'+trainerChallengeLine(t),()=>startTrainer413(t));
      return
    }
  }
}
