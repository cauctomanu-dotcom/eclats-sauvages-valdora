function arenaChallenge(ti){
  const sealIndex=Math.max(1,Math.min(7,Number(ti)||1));
  if((state.seals||[]).includes(sealIndex)){
    dialog('<b>Gardien</b><br>Tu as déjà gagné mon Sceau. Reviens me voir si tu veux parler entraînement.');
    return
  }
  const count=v83GuardianPartySize(sealIndex);
  const base=v83GuardianBaseLevel(sealIndex);
  const guardian={
    id:'guardian_seal_'+sealIndex,
    guardian:true,
    sealIndex,
    look:15,
    level:base,
    dialog:'Gardien du Sceau '+sealIndex,
    zone:state.zone
  };
  dialog('<b>Gardien du Sceau '+sealIndex+'</b><br>'+
    'Pour obtenir ce Sceau, tu dois me vaincre. Mon équipe compte <b>'+count+
    ' créatures</b>, autour du niveau <b>'+base+'</b>. Prépare ton équipe !',
    ()=>startTrainer413(guardian))
}
