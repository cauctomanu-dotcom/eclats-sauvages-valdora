function playerQuestGateMessage(from,to){
  if(!PLAYER_PROGRESSION_LOCK)return null;
  if(from==='town0'&&to==='route0'&&!state.team.length)return 'Le Professeur Aurine t’attend au laboratoire. Choisis d’abord ton premier compagnon.';
  if(from.startsWith('route')){
    const ri=+from.replace('route','');
    if(to==='town'+(ri+1)&&!routeTrainersComplete(ri))return 'Avant de continuer, termine ta quête sur cette route : affronte les dresseurs qui te barrent le chemin.';
  }
  if(from.startsWith('town')){
    const ti=+from.replace('town','');
    if(ti>0&&to==='route'+ti&&!(state.seals||[]).includes(ti))return 'Ta quête dans cette ville n’est pas terminée. Obtiens d’abord son Sceau.';
  }
  return null
}
