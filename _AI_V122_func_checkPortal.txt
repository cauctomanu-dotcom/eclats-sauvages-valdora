function checkPortal(){
  const sc=currentScene();
  for(const ex of (sc.exits||[])){
    if(!pointIn(state.x,state.y,ex))continue;

    if(ex.finalTemple&&!finalTempleAvailableV77()){
      toast(typeof v83CitadelGateText==='function'?v83CitadelGateText():'La Citadelle exige 75 espèces, les 7 Sceaux et la maîtrise de Vol.');
      // repousser légèrement le joueur à l'intérieur de Taronis
      if(ex.side==='east')state.x-=45;
      else if(ex.side==='west')state.x+=45;
      else if(ex.side==='north')state.y+=45;
      else state.y-=45;
      return
    }

    const target=ex.to;
    if(!target||!SCENES[target]){
      toast('Cette liaison n’est pas encore disponible.');
      return
    }

    // En arrivant, on entre depuis le côté opposé du portail franchi.
    transitionWithQuestGate(target,opposite(ex.side));
    return
  }
}

## game/index.html line 9445
checkPortal=function(){
  if(Date.now()<Number(state.v78PortalLock||0))return;
  const sc=currentScene();
  for(const ex of (sc.exits||[])){
    if(!pointIn(state.x,state.y,ex))continue;
    const from=state.zone,target=ex.to;

    if(sc.kind==='town'&&V78_BY_ROUTE[target]){
      const end=ex.v78Enter;
      if(!end){toast('Liaison V78 invalide');return}
      state.zone=target;
      if(!state.discovered.includes(target))state.discovered.push(target);
      v78PlaceOnRoute(target,end);
      state.v78PortalLock=Date.now()+400;
      resetFollowerTrail();zoneTransitionSoundV77();refreshMusicV77();
      toast(SCENES[target].name);logEvent('Arrivée : '+SCENES[target].name+'.');
      setTimeout(()=>maybeSpawnTaron(false),450);
      return;
    }

    if(sc.kind==='route'&&V78_BY_ROUTE[from]&&ex.v78End){
      state.zone=target;
      if(!state.discovered.includes(target))state.discovered.push(target);
      v78PlaceInCity(target,from);
      state.v78PortalLock=Date.now()+400;
      resetFollowerTrail();zoneTransitionSoundV77();refreshMusicV77();
      toast(SCENES[target].name);logEvent('Arrivée : '+SCENES[target].name+'.');
      return;
    }

    // Sanctuaire et autres transitions restent gérés par le système précédent.
    return V78_OLD_CHECK();
  }
}
